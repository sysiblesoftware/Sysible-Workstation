# Sysible Workstation — GRUB theme.
# Dark topographic background, Sysible lockup near the top, and a subtle
# translucent card with a thin green left accent behind the selected entry
# (a sliced pixmap, selbar_*.png) — no saturated "pill". Fonts are the GRUB
# default (no .pf2 needed); the layout + selection bar carry the look.

title-text: ""
desktop-image: "background.png"
# The Sysible lockup is BAKED INTO background.png — it is NOT a separate GRUB
# `+ image` element any more. Why: arm64 GRUB positioned that image element but
# IGNORED its width/height, drawing logo.png at full native size (1040px) — huge,
# overlapping the menu. That is the bug that kept coming back in pixels AND in
# percentages, because the real failure was GRUB not honouring the element's size
# at all. Baked into the background there is NO element for GRUB to mis-size: GRUB
# only scales the background to the screen, its most basic always-present behaviour.
# `crop` = cover (preserve aspect, fill, centre-crop overflow) so the logo never
# distorts; and if a GRUB build ignored scale-method too, the fallback is
# stretch-to-fill, which keeps the logo in the SAME relative place. No code path
# left can render the logo huge or over the menu. (Re-bake after a logo change:
# branding/logo/grub-background-base.png is the clean topo — composite logo.png at
# 30% width, top 13%, centred, and overwrite background.png.)
desktop-image-scale-method: "crop"
desktop-image-h-align: "center"
desktop-image-v-align: "center"
desktop-color: "#0a0d13"
message-color: "#a7b4cc"
message-bg-color: "#0a0d13"

# Boot entries — a centred column pulled up right under the logo (no big empty
# gap). Selected row = subtle card + green left bar.
# Geometry note: the selected pill grows by item_padding on every side, so
# item_spacing MUST stay larger than 2*item_padding or the pill bleeds into the
# next entry (that was the old "one green box over both entries" bug). Here
# item_spacing (20) far exceeds 2*item_padding (12) — always one clean row.
+ boot_menu {
    left = 29%
    width = 42%
    top = 40%
    height = 44%
    item_color = "#8a93a0"
    selected_item_color = "#ffffff"
    item_height = 40
    item_padding = 6
    item_spacing = 20
    selected_item_pixmap_style = "selbar_*.png"
    scrollbar = false
}

# Footer tagline.
+ label {
    top = 92%
    left = 0
    width = 100%
    align = "center"
    color = "#6f7d94"
    text = "Sysible Workstation — Engineering Automation Cloud"
}
